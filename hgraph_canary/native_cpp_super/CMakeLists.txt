cmake_minimum_required(VERSION 3.25)
project(ess_hgraph_native_cpp_superproject LANGUAGES CXX)

if(NOT DEFINED HGRAPH_SOURCE_DIR)
    message(FATAL_ERROR "HGRAPH_SOURCE_DIR is required")
endif()

set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(HGRAPH_BUILD_PYTHON_BINDINGS OFF CACHE BOOL "" FORCE)
set(HGRAPH_ENABLE_PYTHON_USER_NODES OFF CACHE BOOL "" FORCE)
set(HGRAPH_ENABLE_IDE_PYTHON_HEADER_HINTS OFF CACHE BOOL "" FORCE)
set(HGRAPH_USE_PYARROW_ARROW ON CACHE BOOL "" FORCE)
set(HGRAPH_FETCH_DATE ON CACHE BOOL "" FORCE)
set(HGRAPH_FETCH_SIMDJSON ON CACHE BOOL "" FORCE)
set(HGRAPH_ENABLE_PCH OFF CACHE BOOL "" FORCE)
set(HGRAPH_ENABLE_COMPILER_CACHE OFF CACHE BOOL "" FORCE)

get_filename_component(HGRAPH_SOURCE_DIR_ABS "${HGRAPH_SOURCE_DIR}" ABSOLUTE)
get_filename_component(ESS_CANARY_CPP_DIR "${CMAKE_CURRENT_LIST_DIR}/../cpp" ABSOLUTE)

add_subdirectory("${HGRAPH_SOURCE_DIR_ABS}" hgraph)
add_subdirectory("${ESS_CANARY_CPP_DIR}" ess_canary)
